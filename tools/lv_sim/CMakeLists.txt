cmake_minimum_required(VERSION 3.16)
project(deskmate_sim C CXX)
set(CMAKE_C_STANDARD 99)
set(LVGL_DIR "${CMAKE_CURRENT_LIST_DIR}/../../_lv_sim/lvgl" CACHE PATH "LVGL 9 checkout")
if(NOT EXISTS "${LVGL_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Supply -DLVGL_DIR=/path/to/lvgl (version 9)")
endif()
find_package(SDL2 REQUIRED)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl)
target_include_directories(lvgl SYSTEM PUBLIC ${SDL2_INCLUDE_DIRS})
target_link_libraries(lvgl PUBLIC ${SDL2_LIBRARIES})
add_executable(deskmate sim_main.c ../../firmware/desk-mate-app/focus_font_120.c)
target_link_libraries(deskmate PRIVATE lvgl ${SDL2_LIBRARIES})
# Release builds define NDEBUG; the simulator's assertions are the tests.
target_compile_options(deskmate PRIVATE -UNDEBUG)
add_executable(timer_test ../../tests/focus_model_test.c)
target_compile_options(timer_test PRIVATE -UNDEBUG)
add_executable(ble_model_test ../../firmware/test/test_ble_model.c)
target_compile_options(ble_model_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_executable(power_hold_test ../../tests/power_hold_test.c)
target_compile_options(power_hold_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_executable(idle_brightness_test ../../tests/idle_brightness_test.c)
target_compile_options(idle_brightness_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_executable(voice_hosts_test ../../firmware/test/test_voice_hosts.c)
target_compile_options(voice_hosts_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_executable(voice_host_store_test ../../firmware/test/test_voice_host_store.cpp ../../firmware/desk-mate-app/src/bili_voice/voice_hosts.cpp)
target_include_directories(voice_host_store_test PRIVATE ../../firmware/test/host_stubs)
target_compile_features(voice_host_store_test PRIVATE cxx_std_11)
target_compile_options(voice_host_store_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_executable(voice_wire_test ../../firmware/test/test_voice_wire.c)
target_compile_options(voice_wire_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_executable(voice_fsm_timing_test ../../firmware/test/test_voice_fsm_timing.c ../../firmware/desk-mate-app/src/bili_voice/voice_fsm.c)
target_compile_options(voice_fsm_timing_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_executable(adpcm_test ../../firmware/test/test_adpcm.c ../../firmware/desk-mate-app/src/bili_voice/adpcm.c)
target_compile_options(adpcm_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
enable_testing()
add_test(NAME adpcm COMMAND adpcm_test)
add_test(NAME voice_fsm_timing COMMAND voice_fsm_timing_test)
add_test(NAME voice_wire COMMAND voice_wire_test)
add_test(NAME voice_host_store COMMAND voice_host_store_test)
add_test(NAME voice_hosts COMMAND voice_hosts_test)
add_test(NAME power_hold COMMAND power_hold_test)
add_test(NAME idle_brightness COMMAND idle_brightness_test)
add_test(NAME ble_policy COMMAND ble_model_test)
add_executable(ble_voice_test ../../firmware/test/test_ble_voice.c)
target_compile_options(ble_voice_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME ble_voice_protocol COMMAND ble_voice_test)
add_executable(ble_voice_subscription_test ../../firmware/test/test_ble_voice_subscription.cpp)
target_compile_features(ble_voice_subscription_test PRIVATE cxx_std_11)
target_compile_options(ble_voice_subscription_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME ble_voice_subscription COMMAND ble_voice_subscription_test)
add_test(NAME timer COMMAND timer_test)
add_test(NAME swipe_and_ui COMMAND deskmate --test)
set(LVGL8_DIR "" CACHE PATH "Optional LVGL 8.4 checkout to test the hardware renderer")
set(LVGL8_POOL_BYTES 1048576 CACHE STRING "LVGL 8 pool (32768 for actual hardware memory tests)")
if(LVGL8_DIR)
  file(GLOB_RECURSE LV8_SOURCES CONFIGURE_DEPENDS "${LVGL8_DIR}/src/*.c")
  add_library(lvgl8 STATIC ${LV8_SOURCES})
  target_include_directories(lvgl8 PUBLIC "${LVGL8_DIR}")
  target_compile_definitions(lvgl8 PUBLIC LV_CONF_PATH=${CMAKE_CURRENT_LIST_DIR}/lv_conf_v8.h LV_SIM_POOL_BYTES=${LVGL8_POOL_BYTES})
  add_executable(deskmate_v8 sim_main.c ../../firmware/desk-mate-app/focus_font_120.c)
  target_include_directories(deskmate_v8 PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(deskmate_v8 PRIVATE lvgl8 ${SDL2_LIBRARIES})
  add_test(NAME hardware_lvgl8_swipe_and_ui COMMAND deskmate_v8 --test)
  set_tests_properties(hardware_lvgl8_swipe_and_ui PROPERTIES TIMEOUT 20)
endif()
set_tests_properties(timer swipe_and_ui PROPERTIES TIMEOUT 30)
